import { LightningElement, track } from 'lwc';
import getRelatedObjectsAndRecords from '@salesforce/apex/APM_apLWC_MyRecordsController.getRelatedObjectsAndRecords';
import getRelated from '@salesforce/apex/APM_apLWC_MyRecordsController.getAllRelatedRecords';

const PAGE_SIZE = 10;

function formatLabel(apiName) {
    // Remove '__c', '__mdt', and '__r' suffixes, then replace underscores with spaces
    return apiName.replace(/__c$|__mdt$|__r$/i, '').replace(/_/g, ' ');
}

function getFieldLabel(fieldName) {
    // Use the part before the dot if present, then format
    const namePart = fieldName.includes('.') ? fieldName.split('.')[0] : fieldName;
    return formatLabel(namePart);
}



export default class ApmMyRecords extends LightningElement {
    @track relatedData = [];
    @track pages = {};
    @track related = [];
    @track error;
    @track showmodal = false;

    connectedCallback() {
        this.fetchRelatedData();
    }

    fetchRelatedData() {
        getRelatedObjectsAndRecords()
            .then((data) => {
                console.log(JSON.stringify(data))
                    this.relatedData = Object.entries(data).map(([objectName, value]) => ({
                        objectName: objectName,
                        objectLabel: formatLabel(objectName),
                        fields: value.fields
                            .filter(field => field !== 'Id' && field !== 'Name')
                            .map(field => ({
                                apiName: field,
                                label: getFieldLabel(field)
                            })),
                        records: value.records.map(rec => ({
                            name: rec.Name,
                            recordUrl: '/' + rec.Id,
                            raw: rec, // Store the full record for logging
                            fieldValues: value.fields
                                .filter(field => field !== 'Id' && field !== 'Name')
                                .map(fieldName => rec[fieldName] || 'N/A')
                        }))
                    }));
                    this.pages = {};
                    this.relatedData.forEach(obj => {
                        this.pages[obj.objectName] = 1;
                    });
                    this.error = undefined;
                })
                .catch((error) => {
                    this.error = error;
                    this.relatedData = [];
                this.relatedData = Object.entries(data).map(([objectName, value]) => {
                    const hasNameField = value.fields.includes('Name');

                    return {
                        objectName: objectName,
                        objectLabel: formatLabel(objectName),
                        hasNameField,
                        fields: value.fields.filter(field => field!== 'Id' && field !== 'Name')
                            .map(field => ({
                                apiName: field,
                                label: getFieldLabel(field)
                            })),
                        records: value.records.map(rec => ({
                            name: rec.Name,
                            recordUrl: '/' + rec.Id,
                            raw: rec,
                            fieldValues: value.fields
                                .filter(field => field !== 'Id' && field !== 'Name')
                                .map(fieldName => rec[fieldName] || 'N/A')
                        }))
                    };

                });
                this.pages = {};
                this.relatedData.forEach(obj => {
                    this.pages[obj.objectName] = 1;
                });
                this.error = undefined;
            })
            .catch((error) => {
                this.error = error;
                this.relatedData = [];
            });

    }

    get pagedObjects() {
        return this.relatedData.map(obj => {
            const page = this.pages[obj.objectName] || 1;
            const start = (page - 1) * PAGE_SIZE;
            const totalPages = Math.max(1, Math.ceil(obj.records.length / PAGE_SIZE));
            //Changes
            const slicedRecords = obj.records.slice(start, start + PAGE_SIZE);

            const processedPageRecords = slicedRecords.map(record => {
                const hasName = !!record.name;

                return {
                    ...record,
                    fieldValues: record.fieldValues.map((value, index) => {
                        return {
                            value,
                            isLink: !hasName && index === 0
                        };
                    }),
                    hasName
                }
            })

            //End of Changes
            return {
                ...obj,
                currentPage: page,
                // pageRecords: obj.records.slice(start, start + PAGE_SIZE),
                pageRecords: processedPageRecords,
                totalPages,
                isPrevDisabled: page === 1,
                isNextDisabled: page === totalPages,
                resultStats: {
                    start: obj.records.length ? start + 1 : 0,
                    end: Math.min(page * PAGE_SIZE, obj.records.length),
                    total: obj.records.length
                },
                hasNameField: slicedRecords.some(rec => !!rec.name)
            };
        });
    }

    handlePrevPage(event) {
        const objectName = event.target.dataset.objectname;
        const page = this.pages[objectName] || 1;
        if (page > 1) {
            this.pages = { ...this.pages, [objectName]: page - 1 };
        }
    }

    handleNextPage(event) {
        const objectName = event.target.dataset.objectname;
        const obj = this.relatedData.find(o => o.objectName === objectName);
        const totalPages = Math.max(1, Math.ceil(obj.records.length / PAGE_SIZE));
        const page = this.pages[objectName] || 1;
        if (page < totalPages) {
            this.pages = { ...this.pages, [objectName]: page + 1 };
        }
    }

    handlePrintRecord(event) {
        const currentPgno = Number(event.target.dataset.currentpage);
        let currentRecordIndex = Number(event.target.dataset.index);
        if (currentPgno > 1) {
            currentRecordIndex = currentRecordIndex + ((currentPgno * PAGE_SIZE) - 10);
        }
        const recordIndex = currentRecordIndex;
        // console.log('pgno-->',currentPgno);
        // console.log('currentPageSize-->',PAGE_SIZE);
        // console.log('currentrecordIndex-->',currentRecordIndex);
        // console.log('recordIndex-->',recordIndex);
        const objectIndex = event.target.dataset.objectindex;
        const record = this.relatedData[objectIndex].records[recordIndex];

        let id = '';
        if (record.recordUrl) {
            id = record.recordUrl.replace('/', '');
        } else if (record.raw && record.raw.Id) {
            id = record.raw.Id;
        }

        // Fetch both related records and parent records
        Promise.all([
            getRelated({ recordId: id }),
            getRelatedParent({ childRecordId: id })
        ])
            .then(([relatedData, parentData]) => {

                // Utility function to process data objects (relatedData or parentData)
                function processData(data) {
                    return Object.entries(data).map(([objectName, value]) => ({
                        objectName: objectName,
                        objectLabel: formatLabel(objectName),
                        fields: value.fields
                            .filter(field => field !== 'Id' && field !== 'Name')
                            .map(field => ({
                                apiName: field,
                                label: formatLabel(field)
                            })),
                        records: value.records.map(rec => ({
                            name: rec.Name,
                            recordUrl: '/' + rec.Id,
                            displayFields: value.fields
                                .filter(field => field !== 'Id' && field !== 'Name')
                                .map(fieldName => ({
                                    label: getFieldLabel(fieldName),
                                    value: rec[fieldName] || 'N/A'
                                }))
                        }))
                    }));
                }

                // Process Related Data for both parent and child
                const processedRelatedData = processData(relatedData);
                const processedParentData = processData(parentData);
                // Combine related and parent records into a single list
                this.related = [...processedRelatedData, ...processedParentData];
                this.error = undefined;
            })
            .catch((error) => {
                this.error = error;
                this.related = [];
            });

        this.showmodal = true;
    }

    handlemodalclose() {
        this.related = [];
        this.showmodal = false;
    }
}

    /*
    * @description    Processes a list of configuration SObjects to build and execute queries for related data.
    * @param          params - ProcessConfigsParams wrapper
    * @returns        Map<String, Object> containing related data for each object
    */
    private static Map<String, Object> processConfigs(ProcessConfigsParams params) {
        Map<String, Object> relatedData = new Map<String, Object>();
        String objectField = params.isHomePage ? 'Object__c' : 'RelatedObject__c';

        for (SObject config : params.configs) {
            if (isConfigInvalid(config, objectField, params.validateFields)) { continue; }

            List<String> fieldList = parseFields(getFieldValue(config, 'Fields__c'), !params.isHomePage);
            String objectName = getFieldValue(config, objectField);
            String soql;
            
            if (params.isHomePage) {
                HomeObjectsSoqlParams homeParams = new HomeObjectsSoqlParams();
                homeParams.objectName = objectName;
                homeParams.fieldList = fieldList;
                homeParams.ownershipField = getFieldValue(config, 'Ownership__c');
                homeParams.currentUserId = params.currentUserId;
                // --- ADDED: support WhereClause__c ---
                homeParams.customWhereClause = getFieldValue(config, 'WhereClause__c');
                // --- END ADDED ---
                soql = buildHomeObjectsSoql(homeParams);
            } else {
                RelatedRecordsSoqlParams relatedParams = new RelatedRecordsSoqlParams();
                relatedParams.relatedObject = objectName;
                relatedParams.fieldList = fieldList;
                relatedParams.lookupField = getFieldValue(config, 'Lookup__c');
                relatedParams.recordId = params.recordId;
                soql = buildRelatedRecordsSoql(relatedParams);
            }
            
            AddDataToResponseParams recordsToAdd = new AddDataToResponseParams();
            recordsToAdd.dataMap = relatedData;
            recordsToAdd.key = getFieldValue(config, 'MasterLabel');
            recordsToAdd.fieldList = fieldList;
            recordsToAdd.records = executeQuery(soql, objectName);

            addDataToResponse(recordsToAdd);
        }
        return relatedData;
    }

    /*
    * @author         Suyash Patil  
    * @date           9 June 2025
    * @description    Constructs SOQL for retrieving home page objects.
    * @param          params - HomeObjectsSoqlParams
    * @returns        String SOQL query
    */ 
    // --- CHANGED: Enhanced custom where clause handling ---
    private static String buildHomeObjectsSoql(HomeObjectsSoqlParams params) {
        String baseWhere = String.isNotBlank(params.ownershipField)
            ? params.ownershipField + ' = \'' + params.currentUserId + '\''
            : 'OwnerId = \'' + params.currentUserId + '\'';
        String selectClause = QSELECT+' ' + String.join(params.fieldList, ', ') + ' '+QFROM+' ' + params.objectName;

        if (String.isNotBlank(params.customWhereClause)) {
            String customWhere = params.customWhereClause.trim();

            // If clause starts with "WHERE", always add owner condition first then custom clause after AND
            if (customWhere.toLowerCase().startsWith('where')) {
                String afterWhere = customWhere.substring(5).trim();
                return selectClause + ' WHERE ' + baseWhere + ' AND (' + afterWhere + ')';
            }

            // If clause starts with "LIMIT", just add it to default where
            if (customWhere.toLowerCase().startsWith('limit')) {
                return selectClause + ' WHERE ' + baseWhere + ' ' + customWhere;
            }

            // If clause contains "LIMIT" (e.g. condition + limit)
            if (customWhere.toLowerCase().contains('limit')) {
                Integer limitIndex = customWhere.toLowerCase().indexOf('limit');
                String condition = customWhere.substring(0, limitIndex).trim();
                String limitPart = customWhere.substring(limitIndex).trim();
                return selectClause + ' WHERE ' + baseWhere + ' AND (' + condition + ') ' + limitPart;
            }

            // If clause starts with "AND", append as is
            if (customWhere.toLowerCase().startsWith('and')) {
                return selectClause + ' WHERE ' + baseWhere + ' ' + customWhere + ' LIMIT 10';
            }

            // Otherwise, treat it as a condition (append AND)
            return selectClause + ' WHERE ' + baseWhere + ' AND (' + customWhere + ') LIMIT 10';
        }

        // Default
        return selectClause + ' WHERE ' + baseWhere + ' LIMIT 10';
    }

    /*
    * @author         Suyash Patil  
    * @date           9 June 2025
    * @description    Retrieves the parent Id for a child record using the lookup field.
    * @param          lookupField - lookup field API name
    * @param          childObjectName - Child Object Name
    * @param          childRecordId  Child Record Id
    * @returns        Id of parent record or null if not found
    */
    private static Id getParentId(String lookupField, Id childRecordId, String childObjectName) {
        String childQuery = QSELECT+' ' + lookupField + ' '+QFROM+' ' + childObjectName + ' WHERE Id = \'' + childRecordId + '\'';
        List<SObject> childRecords = executeQuery(childQuery, childObjectName);
        return (childRecords != null && !childRecords.isEmpty()) ? (Id)childRecords[0].get(lookupField) : null;
    }

    * @description    Executes a dynamic SOQL query and returns records. Handles exceptions using logger.
    * @param          soql - The SOQL string to execute
    * @param          objectNameForError - Object name for logging errors
    * @returns        List<SObject> query results or null if error
    */
    private static List<SObject> executeQuery(String soql, String objectNameForError) {
        try {
            return Database.query(soql);

            handleException(e,objectNameForError);
            return null; // Simplified error handling
        }
    }

    /*
    * @description    Parses a comma-separated field string to a list, ensures 'Id' is present if required.
    * @param          fieldsStr - Comma-separated fields
    * @param          ensureId - Whether to ensure 'Id' is included
    * @returns        List<String> of field names
    */
    private static List<String> parseFields(String fieldsStr, Boolean ensureId) {
        List<String> fieldList = parseFields(fieldsStr);
        if (ensureId && !fieldList.contains('Id')) {fieldList.add(0, 'Id');}
        return fieldList;
    }

    /*
    * @description    Parses a comma-separated string of field names to a list.
    * @param          fieldsStr - Comma-separated fields
    * @returns        List<String> of field names
    */
    private static List<String> parseFields(String fieldsStr) {
        List<String> fieldList = new List<String>();
        if (String.isBlank(fieldsStr)) {return fieldList;}
        for (String field : fieldsStr.split(',')) {
            String trimmedField = field.trim();
            if (String.isNotBlank(trimmedField) && !fieldList.contains(trimmedField)) {fieldList.add(trimmedField);}
        }
        return fieldList;
    }

    /*
    * @description    Packages queried data into a map for response, with fields and flattened record values.
    * @param          fieldList - List of field API names
    * @param          records - List of SObject records
    * @returns        Map with 'fields' and 'records' keys for response
    */
    private static Map<String, Object> packageDataForResponse(List<String> fieldList, List<SObject> records) {
        Map<String, Object> objectData = new Map<String, Object>();
        objectData.put('fields', fieldList);
        objectData.put('records', flattenRecords(records, fieldList));
        return objectData;
    }

    /*
    * @description    Flattens a list of SObjects into a list of maps containing only the requested fields.
    * @param          records - List of SObjects
    * @param          fieldList - List of field API names
    * @returns        List<Map<String, Object>> for each record
    */
    private static List<Map<String, Object>> flattenRecords(List<SObject> records, List<String> fieldList) {
        List<Map<String, Object>> flatRecords = new List<Map<String, Object>>();
        for (SObject record : records) {
            Map<String, Object> flatRecord = new Map<String, Object>();
            for (String field : fieldList) {flatRecord.put(field, getRelationshipFieldValue(record, field));}
            flatRecords.add(flatRecord);
        }
        return flatRecords;
    }

    /*
    * @description    Gets a field value from an SObject, supporting relationship fields (dot notation).
    * @param          record - SObject record
    * @param          fieldName - Field or relationship path (e.g., 'Owner.Name')
    * @returns        Field value or null if not present
    */
    private static Object getRelationshipFieldValue(SObject record, String fieldName) {
        if (!fieldName.contains('.')) {return record.get(fieldName);}

        List<String> parts = fieldName.split('\\.');
        SObject tempObj = record;
        for (Integer i = 0; i < parts.size() - 1; i++) {
            if (tempObj == null) {return null;}
            tempObj = tempObj.getSObject(parts[i]);
        }
        return tempObj != null ? tempObj.get(parts[parts.size() - 1]) : null;
    }
